pour la supprimer # On se connecte à la base de données 'postgres' et non 'darija_db':

SELECT pg_terminate_backend(pg_stat_activity.pid)
FROM pg_stat_activity
WHERE pg_stat_activity.datname = 'darija_db'
  AND pid <> pg_backend_pid();
DROP DATABASE darija_db;
CREATE DATABASE darija_db;
\q
