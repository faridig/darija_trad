pour augmenter le niveau de hadrware de l'api ia : 

# --- Variables ---                                                                                                                     
RESOURCE_GROUP="RG_FIGOUTI"
PLAN_NAME="plan-darija-api"

echo "Mise à niveau du plan de service '${PLAN_NAME}' vers P3V2..."

az appservice plan update \
  --name "$PLAN_NAME" \
  --resource-group "$RESOURCE_GROUP" \
  --sku P3V2

echo "✅ Mise à niveau vers P3V2 terminée."
echo "L'application va maintenant être redémarrée sur une machine plus puissante."

pour le plan moins cher en cas de non utilisation  :

az appservice plan update \
  --name "plan-darija-api" \
  --resource-group "RG_FIGOUTI" \
  --sku B1
